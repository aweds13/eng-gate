- 解析顺序（命中即停并记桶；多份全列）：
  **A** 文件名含台账号（**字面子串**，不做边界严格匹配）→

  **D** 台账 `text:复盘/…` 证据指针所指文件（共享文档，只列路径）→

## 六、生成脚本副本（证据；运行于 `%TEMP%/card076/gen_index.py`，仓库根执行，不进仓库）

`tools/` 与 `scripts/` 是本卡禁区，脚本按卡 074 的先例放 TEMP；`analyze` 只读盘点、`gen` 覆盖索引。

```python
# -*- coding: utf-8 -*-
"""卡076 (T-G171)：台账号 -> 复盘文档索引【随波次重算】（唯一产物：覆盖 复盘/索引-台账号到复盘.md）。
规格：gates/diag/卡076-索引随波次重算.md；底座 c2fed5e。
算法逐字沿用卡074（T-G168）首版脚本（见 复盘/台账号到复盘索引-T-G168.md §五），只改头部产物文字。
解析顺序（命中即停并记桶；多份则全列）：
  A 文件名含台账号（子串，卡口径）
  B 正文里以该号开头有 标题/加粗/引用 的那一份（★ 不放宽到列表前缀/任意提及）
  C 台账 evidence 的 file:复盘/… 指针
  D 台账 evidence 的 text:复盘/… 指针所指文件（只列路径）
  E 都没有 -> ★ 未定位（如实列出；备注给「仅提及」线索的透明标注）
用法（必须在仓库根运行）：
  python gen_index.py analyze    # 只读盘点
  python gen_index.py gen        # 生成 复盘/索引-台账号到复盘.md
"""
import sys, re, tomllib, pathlib

ROOT = pathlib.Path.cwd()
assert (ROOT / 'docs' / 'traceability.toml').exists(), '必须在仓库根运行'
LEDGER = ROOT / 'docs' / 'traceability.toml'
FU = ROOT / '复盘'
ID_RE = re.compile(r'^T-G\d+[a-z]?$', re.IGNORECASE)
# B 桶：行首 = 标题(#)/加粗(**)/引用(>)（引用内可直接跟加粗），随后紧跟台账号
BODY_PAT = lambda ident: re.compile(
    r'^\s*(?:#{1,6}\s*|>\s*(?:\*\*\s*)?|\*\*\s*)' + re.escape(ident) + r'(?![A-Za-z0-9])')
# E 桶备注用：宽松「行首提及」（含裸行首/列表/反引号），仅作透明线索，不作定位
LOOSE_PAT = lambda ident: re.compile(
    r'^[\s>#*`\-]*' + re.escape(ident) + r'(?![A-Za-z0-9])')


def load_ledger():
    d = tomllib.loads(LEDGER.read_text(encoding='utf-8'))
    return d['entries']


def loose_hits(ident, mds):
    pat = LOOSE_PAT(ident)
    out = []
    for p in mds:
        try:
            lines = p.read_text(encoding='utf-8').splitlines()
        except Exception:
            continue
        for j, line in enumerate(lines):
            if pat.match(line):
                out.append(p.relative_to(ROOT).as_posix() + ':' + str(j + 1))
                break
    return out


def main():
    mode = sys.argv[1] if len(sys.argv) > 1 else 'analyze'
    entries = load_ledger()
    all_ids = list(entries)
    tg_ids = [i for i in all_ids if ID_RE.match(i)]
    mds = sorted(FU.rglob('*.md'))
    texts = {}
    for p in mds:
        try:
            texts[p] = p.read_text(encoding='utf-8')
        except Exception:
            texts[p] = ''

    # 基线读数
    n_ev_file = 0
    n_ev_file_fupan = 0
    broken = 0
    for ident, e in entries.items():
        for ev in (e.get('evidence') or []):
            if isinstance(ev, str) and ev.startswith('file:'):
                n_ev_file += 1
                if ev[5:].strip().startswith('复盘/'):
                    n_ev_file_fupan += 1
                if not (ROOT / ev[5:].strip()).exists():
                    broken += 1
    thin = []
    for p in mds:
        st = p.stat()
        nonempty = sum(1 for l in texts[p].splitlines() if l.strip())
        if st.st_size < 1200 or nonempty < 15:
            thin.append(p.name)

    print(f'复盘 md 份数: {len(mds)}')
    print(f'台账条数: {len(all_ids)}')
    print(f'file: 证据指针: {n_ev_file} 条（其中 file:复盘/ {n_ev_file_fupan} 条），解析不到 = {broken}')
    print(f'偏薄/占位 (<1.2KB 或 <15 非空行) = {len(thin)} 份: {thin}')
    print(f'台账号总数 T-G*: {len(tg_ids)}')

    # 桶 A：文件名子串
    fname_map = {}
    for p in mds:
        for ident in tg_ids:
            if ident.lower() in p.name.lower():
                fname_map.setdefault(ident, []).append(p)

    rows = []   # (ident, bucket, [relpath], remark)
    for ident in tg_ids:
        ev = entries[ident].get('evidence') or []
        files_f = [e[5:].strip() for e in ev
                   if isinstance(e, str) and e.startswith('file:复盘/')]
        files_t = sorted({e[5:].split('::')[0].strip() for e in ev
                          if isinstance(e, str) and e.startswith('text:复盘/')})
        if ident in fname_map:
            rows.append((ident, 'A', [p.relative_to(ROOT).as_posix() for p in fname_map[ident]], ''))
            continue
        hits_b = []
        pat_b = BODY_PAT(ident)
        for p in mds:
            for line in texts[p].splitlines():
                if pat_b.match(line):
                    hits_b.append(p)
                    break
        if hits_b:
            rows.append((ident, 'B', [p.relative_to(ROOT).as_posix() for p in hits_b], '正文标题/加粗/引用行首命中'))
            continue
        if files_f:
            rows.append((ident, 'C', files_f, '台账 file:复盘/ 指针'))
            continue
        if files_t:
            rows.append((ident, 'D', files_t, '台账 text:复盘/ 指针（共享文档，索引只列路径）'))
            continue
        lh = loose_hits(ident, mds)
        if lh:
            remark = ('台账证据只指向 src/docs/tools/tests；复盘正文仅他卡文件内提及（按卡 §2.1.2 口径不计定位）：'
                      + '、'.join('`' + h + '`' for h in lh))
        else:
            remark = '台账证据只指向 src/docs/tools/tests'
        rows.append((ident, 'E', [], remark))

    from collections import Counter
    cnt = Counter(b for _, b, _, _ in rows)
    e_ids = [i for i, b, _, _ in rows if b == 'E']
    print(f"分桶：A={cnt.get('A',0)} B={cnt.get('B',0)} C={cnt.get('C',0)} D={cnt.get('D',0)} E={cnt.get('E',0)}")
    print(f'未定位（E 桶）：{e_ids}')

    if mode == 'analyze':
        for ident, b, files, remark in rows:
            if b != 'A':
                print(f'  {ident} [{b}] {files} {remark}')
        return

    if mode == 'gen':
        out = ROOT / '复盘' / '索引-台账号到复盘.md'
        L = []
        L.append('# 台账号 → 复盘文档索引')
        L.append('')
        L.append('> 由脚本现算生成（卡 076 · T-G171 **随波次重算**；底座 `c2fed5e`）：读 `docs/traceability.toml`（只读）+ 扫 `复盘/**/*.md`。')
        L.append('> **本索引随波次重算（见卡 076）**：首版是卡 074（T-G168）按底座 `b90c4e4` 现算的 163 行快照；')
        L.append('> 卡 076 先按集成后台账的 172 个 `T-G*` 重算（补上 R-I10 抓出的 `T-G162…T-G170` 九个），')
        L.append('> 因「追加本卡 `T-G171` 条目后必须再算一次」（正是 §24.11 的同波口径），本版覆盖台账全部 **173** 个 `T-G*`（含 `T-G171` 自己）。')
        L.append('> 解析顺序（命中即停并记桶；多份全列）：**A** 文件名含台账号 → **B** 正文以该号开头有标题/加粗/引用 →')
        L.append('> **C** 台账 `file:复盘/…` 指针 → **D** 台账 `text:复盘/…` 指针（多为共享文档，索引只列路径）→')
        L.append('> **E** 都没有 = ★ 未定位（如实标注，不硬塞）。')
        L.append('> ★ 索引只放路径，不复制正文（避免两份真源）。算法与首版脚本见 `复盘/台账号到复盘索引-T-G168.md` §五；')
        L.append('> 本次重算记录（脚本副本/读数/diff）见 `复盘/索引重算与规矩-T-G171.md`。既有复盘文件零改动。')
        L.append('')
        L.append('| 台账号 | 状态 | 复盘文档（相对路径） | 备注 |')
        L.append('|---|---|---|---|')
        for ident, b, files, remark in rows:
            label = {'A': 'A·文件名', 'B': 'B·正文行首', 'C': 'C·file:指针',
                     'D': 'D·text:指针', 'E': 'E·未定位'}[b]
            cell = '；'.join('`' + f + '`' for f in files) if files else '★ 未定位'
            L.append('| `' + ident + '` | ' + label + ' | ' + cell + ' | ' + remark + ' |')
        L.append('')
        L.append('## 数字 → 出处')
        L.append('')
        L.append(f'- 台账条目总数：{len(all_ids)}（`docs/traceability.toml` [entries]，现算）')
        L.append(f'- 台账号（`T-G*`）总数：**{len(tg_ids)}**')
        L.append(f"- 桶 A（文件名含台账号）：**{cnt.get('A',0)}**")
        L.append(f"- 桶 B（正文标题/加粗/引用行首）：**{cnt.get('B',0)}**")
        L.append(f"- 桶 C（台账 file:复盘/ 指针）：**{cnt.get('C',0)}**")
        L.append(f"- 桶 D（台账 text:复盘/ 指针）：**{cnt.get('D',0)}**")
        L.append(f"- 桶 E（未定位）：**{cnt.get('E',0)}**（{'、'.join(e_ids)}）")
        L.append(f'- 复盘 md 总份数：{len(mds)}；`file:` 证据指针 {n_ev_file} 条、断链 {broken} 条、偏薄/占位 {len(thin)} 份')
        L.append('')
```
